def materialization_uuid_candidate():
    source_file = Path(promotion_paging.__file__)
    source_sha = hashlib.sha256(source_file.read_bytes()).hexdigest()
    assert source_sha == "79665c4cd351f668f4f8378ead3ce514b6fdd6491f737e1b6c8294b31913009e"
    source = inspect.getsource(promotion_paging._snapshot_order_population)
    needle = "rights.iterator(chunk_size=500)"
    assert source.count(needle) == 1
    candidate = source.replace(needle, "rights")
    loop = "for pk, kind, bumped_at in rights:"
    assert candidate.count(loop) == 1 and candidate.count("str(pk)") == 4
    candidate = candidate.replace(loop, "for raw_pk, kind, bumped_at in rights:\n        pk = str(raw_pk)")
    candidate = candidate.replace("str(pk)", "pk")
    namespace = vars(promotion_paging).copy()
    exec(compile(candidate, "<beta-rights-materialization-uuid-candidate>", "exec"), namespace)
    return namespace["_snapshot_order_population"], source_sha, hashlib.sha256(candidate.encode()).hexdigest()


def test_rights_materialization_uuid_balanced_roi(request, settings, monkeypatch, record_property):
    run_id = os.environ.get("LOCAL_CI_ISOLATED_BROKER_RUN")
    assert run_id and connection.vendor == "postgresql" and not settings.PRODUCTION
    assert not publication_state.READ_REPAIR
    gc_policy = (gc.isenabled(), gc.get_threshold(), gc.get_debug())
    assert gc_policy[0] and gc_policy[2] == 0
    folder = settings.BASE_DIR / "artifacts/local-ci" / run_id
    runner = json.loads((folder / "runner.json").read_text())
    container = os.environ["LOCAL_CI_POSTGRES_CONTAINER"]
    assert container == "web-modelos-ci-pg-" + run_id
    inspection = json.loads(subprocess.check_output(["docker", "inspect", container], text=True))[0]
    settings.PROMOTION_TEST_MODE = True
    dataset = prepare_promotional_load(request, 10000)
    publication_state.refresh_all()
    assert not PublicationDirty.objects.exists()
    original = promotion_paging._snapshot_order_population
    proposed, source_sha, candidate_sha = materialization_uuid_candidate()
    chosen = {"function": original}

    def dispatch(*args, **kwargs):
        observation = chosen.get("observation")
        if observation is None:
            return chosen["function"](*args, **kwargs)
        chosen["commercial_active"] = True
        allocation_start = tracemalloc.get_traced_memory()[0]
        tracemalloc.reset_peak()
        start, cpu = time.perf_counter(), time.thread_time()
        try:
            return chosen["function"](*args, **kwargs)
        finally:
            observation["commercial_wall_seconds"] = time.perf_counter() - start
            observation["commercial_thread_cpu_seconds"] = time.thread_time() - cpu
            allocated, peak = tracemalloc.get_traced_memory()
            observation["commercial_python_peak_increment_bytes"] = peak - allocation_start
            observation["commercial_python_retained_increment_bytes"] = allocated - allocation_start
            chosen["commercial_active"] = False

    monkeypatch.setattr(promotion_paging, "_snapshot_order_population", dispatch)
    client = request.getfixturevalue("directory")

    def read(path):
        response = client.get(path)
        assert response.status_code == 200 and response["Cache-Control"] == "no-store"
        assert response.wsgi_request._publication_route["route"] == "state"
        page = response.context["page"]
        return (hashlib.sha256(response.content).hexdigest(), page.paginator.count,
                tuple(str(row.pk) for row in page.object_list),
                tuple(card["sponsorship"] for card in response.context["cards"]))

    records = []
    signatures = {}
    for route, path in [("directory_es", "/es/"), ("search_en", "/en/?q=synthetic")]:
        for block, order in enumerate(itertools.permutations("ABC")):
            for arm in order:
                chosen["function"] = proposed if arm == "C" else original
                connection.close()
                signature = read(path)  # One warm read per newly opened backend, outside controls.
                signatures.setdefault(route, signature)
                assert signature == signatures[route] and len(signature[2]) == 24 and any(signature[3])
                with connection.cursor() as cursor:
                    cursor.execute("SELECT pg_backend_pid(), current_setting('cursor_tuple_fraction')")
                    sql_pid, cursor_fraction = cursor.fetchone()
                identity = owned_backend(runner, inspection, sql_pid)

                def snapshot():
                    state = backend_snapshot(identity)
                    sched = list(map(int, Path(f"/proc/{identity['pid']}/schedstat").read_text().split()))
                    assert len(sched) == 3 and sched[0] > 0
                    return state | {"runtime_ns": sched[0], "wait_ns": sched[1]}

                before = snapshot()
                samples = []
                for _ in range(3):
                    started, cpu = time.perf_counter(), time.thread_time()
                    current = read(path)
                    samples.append({"wall_seconds": time.perf_counter() - started,
                                    "thread_cpu_seconds": time.thread_time() - cpu})
                    assert current == signature
                after = snapshot()
                assert after["runtime_ns"] >= before["runtime_ns"]
                with connection.cursor() as cursor:
                    cursor.execute("SELECT pg_backend_pid(), current_setting('cursor_tuple_fraction')")
                    assert cursor.fetchone() == (sql_pid, cursor_fraction)
                assert not PublicationDirty.objects.exists()
                records.append({"route": route, "block": block, "order": "".join(order), "arm": arm,
                                "samples": samples, "backend_cpu_seconds": (
                                    after["runtime_ns"] - before["runtime_ns"]) / 1e9,
                                "backend_wait_seconds": (after["wait_ns"] - before["wait_ns"]) / 1e9,
                                "cursor_tuple_fraction": cursor_fraction,
                                "backend_identity": identity, "before": before, "after": after})
    observations, plans = [], {}
    for route, path in [("directory_es", "/es/"), ("search_en", "/en/?q=synthetic")]:
        captured = []
        for arm in "AC":
            chosen["function"] = proposed if arm == "C" else original
            observation = {"route": route, "arm": arm, "execute_calls": 0,
                           "execute_seconds": 0.0, "stream_fetch_calls": 0,
                           "stream_fetch_seconds": 0.0, "stream_rows": 0, "commercial_uuid_str_calls": 0}
            chosen["observation"] = observation
            original_fetch = ServerSideCursor.fetchmany
            original_uuid_str = uuid.UUID.__str__

            def uuid_str(value):
                if chosen.get("commercial_active"):
                    observation["commercial_uuid_str_calls"] += 1
                return original_uuid_str(value)

            def fetch(cursor, *args, **kwargs):
                start = time.perf_counter()
                rows = original_fetch(cursor, *args, **kwargs)
                if chosen.get("commercial_active"):
                    observation["stream_fetch_calls"] += 1
                    observation["stream_fetch_seconds"] += time.perf_counter() - start
                    observation["stream_rows"] += len(rows)
                return rows

            def execute(query_execute, sql, params, many, context):
                start = time.perf_counter()
                try:
                    return query_execute(sql, params, many, context)
                finally:
                    if chosen.get("commercial_active"):
                        observation["execute_calls"] += 1
                        observation["execute_seconds"] += time.perf_counter() - start
                        if arm == "A":
                            captured.append((sql, params))

            assert not tracemalloc.is_tracing()
            try:
                tracemalloc.start()
                with monkeypatch.context() as patch, connection.execute_wrapper(execute):
                    patch.setattr(ServerSideCursor, "fetchmany", fetch)
                    patch.setattr(uuid.UUID, "__str__", uuid_str)
                    assert read(path) == signatures[route]
            finally:
                tracemalloc.stop()
                chosen["observation"] = None
            assert observation["execute_calls"] == 1
            if arm == "A":
                assert observation["stream_fetch_calls"] > 0 and observation["stream_rows"] > 0
            else:
                assert observation["stream_fetch_calls"] == observation["stream_rows"] == 0
            observations.append(observation)
        original_calls, candidate_calls = [o["commercial_uuid_str_calls"] for o in observations[-2:]]
        assert original_calls > 0 and 0 < candidate_calls <= original_calls / 2
        assert len(captured) == 1
        sql, params = captured[0]
        assert isinstance(sql, str)
        plans[route] = {}
        with transaction.atomic(), connection.cursor() as cursor:
            cursor.execute("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ READ ONLY")
            for label, prefix in [("ordinary_select", ""), ("cursor", (
                "DECLARE beta_rights_plan NO SCROLL CURSOR WITHOUT HOLD FOR "))]:
                cursor.execute("EXPLAIN (FORMAT JSON) " + prefix + sql, params)
                plan = cursor.fetchone()[0]
                plans[route][label] = json.loads(plan) if isinstance(plan, str) else plan
    comparisons = {}
    for route in signatures:
        selected = [row for row in records if row["route"] == route]
        metrics = {}
        for arm in "ABC":
            rows = [r for r in selected if r["arm"] == arm]
            metrics[arm] = {"wall": statistics.median(s["wall_seconds"] for r in rows for s in r["samples"]),
                            "app_cpu": statistics.median(s["thread_cpu_seconds"] for r in rows for s in r["samples"]),
                            "pg_cpu_per_read": sum(r["backend_cpu_seconds"] for r in rows) / 18}
        stable = all(abs(metrics["A"][key] / metrics["B"][key] - 1) <= 0.10 for key in metrics["A"])
        reference = {key: (metrics["A"][key] + metrics["B"][key]) / 2 for key in metrics["A"]}
        ratios = {key: metrics["C"][key] / value for key, value in reference.items()}
        nonregressing = 0
        for block in range(6):
            group = {r["arm"]: statistics.median(s["wall_seconds"] for s in r["samples"])
                     for r in selected if r["block"] == block}
            nonregressing += group["C"] <= (group["A"] + group["B"]) / 2
        comparisons[route] = {"metrics": metrics, "ratios_C_to_AB": ratios, "AA_stable": stable,
                              "nonregressing_blocks": nonregressing,
                              "eligible": stable and ratios["wall"] <= 0.90
                              and ratios["app_cpu"] <= 1.05 and ratios["pg_cpu_per_read"] <= 1.05
                              and nonregressing >= 4}
    assert len(records) == 36 and sum(len(r["samples"]) for r in records) == 108
    assert gc_policy == (gc.isenabled(), gc.get_threshold(), gc.get_debug())
    result = {"run_id": run_id, "result": "EXPERIMENT_COMPLETE", "count": dataset["published"],
              "rights": dataset["promotions"]["count"], "source_sha256": source_sha,
              "candidate_sha256": candidate_sha, "records": records, "comparisons": comparisons,
              "gc_policy_unchanged": gc_policy,
              "separate_fetch_observations": observations, "planning_only_after_controls": redact(plans),
              "eligible": all(c["eligible"] for c in comparisons.values()),
              "requests": 148, "warm_reads": 36, "measured_reads": 108, "observed_reads": 4,
              "limits": ["Candidate materializes narrow rights and reuses one UUID text per right within the request",
                         "Same authority, population, ordering and native page guard; exact bytes/IDs/count/labels",
                         "A/B identical; six balanced permutations, new owned backend and one warm read per arm",
                         "App thread CPU and identified PG backend schedstat outside control boundaries",
                         "No profiler/SQL wrapper inside controls, normal GC, host shared",
                         "Four separate observed reads attribute commercial execute versus stream fetch",
                         "Tracemalloc only in observed reads, commercial Python peak is not service RSS",
                         "Planning-only SELECT/DECLARE comparison after controls, no runtime/CPU inferred",
                         "Owned local sequential experiment, not TCP/SLO or full host memory measurement",
                         "PASS pytest means experiment integrity; eligible false rejects adoption"]}
    (folder / "beta-rights-materialization-uuid-roi.json").write_text(json.dumps(result, indent=2) + "\n")
    record_property("beta_rights_materialization_uuid_eligible", result["eligible"])
    record_property("beta_rights_materialization_uuid_comparisons", json.dumps(comparisons))
