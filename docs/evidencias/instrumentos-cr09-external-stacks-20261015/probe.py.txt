import json
import os
import subprocess
import sys
from pathlib import Path
from external_stacks import ExternalStacks
out = Path(__file__).resolve().parent
cg = Path('/proc/self/cgroup').read_text()
assert 'cr09-external-stacks-probe-' in cg
assert sorted(os.sched_getaffinity(0)) == [0, 1, 2, 3]
observer = ExternalStacks.__new__(ExternalStacks)
observer.cg = cg
pressure = observer.pressure()
assert pressure['controller_cpu_pressure'] is None
child = subprocess.Popen([sys.executable, '-c', 'import time; time.sleep(10)'])
try:
    argv = ['/home/felidae/.local/share/web-citas-diagnostics/venv/bin/py-spy', 'record', '--pid', str(child.pid),
            '--idle', '--nonblocking', '--duration', '1', '--rate', '100', '--format', 'speedscope',
            '--output', str(out/'probe-speedscope.json')]
    result = subprocess.run(argv, capture_output=True, text=True, timeout=10)
    (out/'probe.log').write_text(result.stdout+result.stderr)
    assert result.returncode == 0, result.stderr
    profile = json.loads((out/'probe-speedscope.json').read_text())
    assert profile['profiles'] and profile['shared']['frames']
finally:
    child.terminate()
    child.wait(timeout=5)
assert not Path('/proc/'+str(child.pid)).exists()
(out/'probe-receipt.json').write_text(json.dumps({'result':'PASS','child_terminal':True,'pressure':pressure,
                                                'profile_count':len(profile['profiles'])},indent=2))
